# CI for web-security-labs
#
# Least privilege by default: the workflow needs to read the repository and run
# tests, and nothing more. `permissions: contents: read` is set at the top level
# so that no job can accidentally escalate to write access.
#
# Third-party actions are pinned to a full commit SHA rather than a tag, so a
# moved tag cannot silently change what runs in this repository.

name: CI

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

permissions:
  contents: read
